#include <tunables/global>

profile ha_db_toolbox flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # The web API listens locally and calls the Supervisor/Core proxies.
  network inet stream,
  network inet6 stream,

  # The entrypoint drops from root to the dedicated runtime user.
  capability chown,
  # Required only while the root bootstrap initializes the app-owned writable
  # mappings. The service immediately drops to UID/GID 10001 afterwards.
  capability dac_override,
  capability setgid,
  capability setuid,

  # Docker and the Supervisor may stop the process gracefully.
  signal (receive),
  signal (send) set=(kill,term,int,hup,cont),

  # Python runtime and installed application.
  /usr/local/bin/ r,
  /usr/local/bin/python* rix,
  /usr/local/bin/uvicorn rix,
  /usr/local/lib/** mr,
  /usr/lib/** mr,
  /lib/** mr,
  /app/ r,
  /app/** r,

  # Explicitly mapped application data only. Home Assistant configuration is
  # not mounted; all HA data is read through documented Supervisor/Core APIs.
  /config/{,**} rwk,
  /data/{,**} rwk,

  # DEV-HA test build: read-only probe of the Recorder SQLite file. No other
  # file in the Home Assistant configuration directory is accessible.
  /homeassistant/ r,
  /homeassistant/home-assistant_v2.db{,-wal,-shm} rmk,

  # Runtime discovery, DNS, entropy, and temporary files.
  /etc/group r,
  /etc/host.conf r,
  /etc/hosts r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/resolv.conf r,
  /etc/ssl/openssl.cnf r,
  /etc/ssl/certs/ca-certificates.crt r,
  /proc/{,**} r,
  /sys/devices/system/cpu/{,**} r,
  /dev/null rw,
  /dev/urandom r,
  /tmp/{,**} rwk,
}
